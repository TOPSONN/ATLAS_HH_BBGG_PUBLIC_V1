cmake_minimum_required(VERSION 3.20)
project(atlas_hh_pythia8_shower LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PYTHIA8_ROOT "" CACHE PATH "Verified isolated Pythia8 prefix")
set(HEPMC3_ROOT "" CACHE PATH "Verified isolated HepMC3 prefix")
find_path(PYTHIA8_INCLUDE_DIR Pythia8/Pythia.h PATHS "${PYTHIA8_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(PYTHIA8_LIBRARY pythia8 PATHS "${PYTHIA8_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
find_path(HEPMC3_INCLUDE_DIR HepMC3/GenEvent.h PATHS "${HEPMC3_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(HEPMC3_LIBRARY HepMC3 PATHS "${HEPMC3_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
find_package(ZLIB REQUIRED)
set(CMAKE_BUILD_RPATH "${PYTHIA8_ROOT}/lib;${HEPMC3_ROOT}/lib")
set(CMAKE_INSTALL_RPATH "${PYTHIA8_ROOT}/lib;${HEPMC3_ROOT}/lib")
foreach(target shower_lhef validate_hepmc3)
  add_executable(${target} src/${target}.cxx)
  target_include_directories(${target} PRIVATE include "${PYTHIA8_INCLUDE_DIR}" "${HEPMC3_INCLUDE_DIR}")
  target_link_libraries(${target} PRIVATE "${PYTHIA8_LIBRARY}" "${HEPMC3_LIBRARY}" ZLIB::ZLIB ${CMAKE_DL_LIBS})
  target_compile_definitions(${target} PRIVATE GZIP)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_executable(test_sha256 tests/test_sha256.cxx)
  target_include_directories(test_sha256 PRIVATE include)
  add_test(NAME exact_event_sha256 COMMAND test_sha256)
  add_executable(test_photon_origin tests/test_photon_origin.cxx)
  target_include_directories(test_photon_origin PRIVATE include)
  add_test(NAME photon_origin_api_fixtures COMMAND test_photon_origin)
  add_executable(test_particle_data tests/test_particle_data.cxx)
  target_include_directories(test_particle_data PRIVATE "${PYTHIA8_INCLUDE_DIR}")
  target_link_libraries(test_particle_data PRIVATE "${PYTHIA8_LIBRARY}" ZLIB::ZLIB ${CMAKE_DL_LIBS})
  add_test(NAME heavy_hadron_particle_data_api COMMAND test_particle_data "${PYTHIA8_ROOT}/share/Pythia8/xmldoc")
  add_executable(test_lha_source_cursor tests/test_lha_source_cursor.cxx)
  target_include_directories(test_lha_source_cursor PRIVATE include "${PYTHIA8_INCLUDE_DIR}")
  target_compile_definitions(test_lha_source_cursor PRIVATE GZIP)
  target_link_libraries(test_lha_source_cursor PRIVATE "${PYTHIA8_LIBRARY}" ZLIB::ZLIB ${CMAKE_DL_LIBS})
  add_test(NAME actual_lhe_reader_cursor_failure_api COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_lha_source_cursor.py" "$<TARGET_FILE:test_lha_source_cursor>"
    "${PYTHIA8_ROOT}/share/Pythia8/xmldoc")
  add_executable(test_string_region_numerics tests/test_string_region_numerics.cxx)
  target_include_directories(test_string_region_numerics PRIVATE "${PYTHIA8_INCLUDE_DIR}")
  target_link_libraries(test_string_region_numerics PRIVATE "${PYTHIA8_LIBRARY}" ZLIB::ZLIB ${CMAKE_DL_LIBS})
  add_test(NAME upstream_degenerate_string_region_diagnosis COMMAND test_string_region_numerics)
endif()
