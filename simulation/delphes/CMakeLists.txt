cmake_minimum_required(VERSION 3.20)
project(atlas_hh_delphes_analysis LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ROOT 6.40.04 EXACT CONFIG REQUIRED COMPONENTS Core RIO Tree Hist Physics EG)
if(NOT ROOT_CXX_STANDARD STREQUAL "20")
  message(FATAL_ERROR "Requires actual ROOT built with C++20")
endif()
set(DELPHES_ROOT "" CACHE PATH "Verified isolated Delphes prefix")
set(DELPHES_SOURCE_DIR "" CACHE PATH "Verified external Delphes source headers")
find_library(DELPHES_LIBRARY Delphes PATHS "${DELPHES_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
find_path(DELPHES_INCLUDE classes/DelphesClasses.h PATHS "${DELPHES_ROOT}/include" "${DELPHES_SOURCE_DIR}" NO_DEFAULT_PATH REQUIRED)
find_path(FASTJET_INCLUDE fastjet/ClusterSequence.hh PATHS "${DELPHES_SOURCE_DIR}/external" "${DELPHES_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
add_executable(analyze_delphes src/analyze_delphes.cxx)
target_include_directories(analyze_delphes PRIVATE include ../pythia8/include "${DELPHES_INCLUDE}" "${FASTJET_INCLUDE}")
target_link_libraries(analyze_delphes PRIVATE "${DELPHES_LIBRARY}" ROOT::Core ROOT::RIO ROOT::Tree ROOT::Hist ROOT::Physics ROOT::EG ${CMAKE_DL_LIBS})
set_target_properties(analyze_delphes PROPERTIES BUILD_RPATH "${DELPHES_ROOT}/lib;${ROOT_LIBRARY_DIR}" INSTALL_RPATH "${DELPHES_ROOT}/lib;${ROOT_LIBRARY_DIR}")
target_compile_options(analyze_delphes PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_analyzer_physics tests/test_analyzer_physics.cxx)
  target_include_directories(test_analyzer_physics PRIVATE include)
  add_test(NAME delphes_analysis_math_and_assignment COMMAND test_analyzer_physics)
endif()
