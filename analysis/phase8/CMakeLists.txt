cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(atlas_hh_phase8 LANGUAGES CXX)
 include(CTest)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
 message(FATAL_ERROR "Phase8 requires an out-of-source build")
endif()
find_package(ROOT 6.40.04 EXACT CONFIG REQUIRED COMPONENTS Core RIO Tree Hist Physics Matrix RooFit RooStats)
if(NOT ROOT_CXX_STANDARD STREQUAL "20")
 message(FATAL_ERROR "Phase8 requires actual ROOT C++20")
endif()
if(NOT CMAKE_CXX_COMPILER MATCHES "conda-linux-gnu")
 message(FATAL_ERROR "Use the verified existing conda C++ compiler")
endif()
set(DELPHES_ROOT "" CACHE PATH "Previously verified Delphes install prefix")
find_library(PHASE8_DELPHES_LIBRARY Delphes PATHS "${DELPHES_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
find_path(PHASE8_DELPHES_INCLUDE classes/DelphesClasses.h PATHS "${DELPHES_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_path(PHASE8_JSON_INCLUDE nlohmann/json.hpp PATHS ${ROOT_INCLUDE_DIRS} NO_DEFAULT_PATH REQUIRED)
get_filename_component(PHASE8_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(atlas_phase8
 src/Phase8Provenance.cxx src/Phase8Reader.cxx src/Phase8Selection.cxx
 src/Phase8Reconstruction.cxx src/Phase8TruthMatching.cxx src/Phase8Histogram.cxx)
target_include_directories(atlas_phase8 PUBLIC include "${PHASE8_REPO}/simulation/delphes/include" "${PHASE8_REPO}/simulation/pythia8/include" "${PHASE8_DELPHES_INCLUDE}" "${PHASE8_JSON_INCLUDE}")
target_link_libraries(atlas_phase8 PUBLIC "${PHASE8_DELPHES_LIBRARY}" ROOT::Core ROOT::RIO ROOT::Tree ROOT::Hist ROOT::Physics ROOT::Matrix ROOT::RooFit ROOT::RooStats)
target_compile_features(atlas_phase8 PUBLIC cxx_std_20)
target_compile_options(atlas_phase8 PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(atlas_phase8 PROPERTIES BUILD_RPATH "${DELPHES_ROOT}/lib;${ROOT_LIBRARY_DIR}")
foreach(app phase8_analyze phase8_validate)
 add_executable(${app} apps/${app}.cxx)
 target_link_libraries(${app} PRIVATE atlas_phase8)
 set_target_properties(${app} PROPERTIES BUILD_RPATH "${DELPHES_ROOT}/lib;${ROOT_LIBRARY_DIR}")
endforeach()
# Statistical modules are supplied as an independent source-owned component.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/Phase8Statistics.cxx")
 target_sources(atlas_phase8 PRIVATE src/Phase8Statistics.cxx src/Phase8Weights.cxx src/Phase8Kappa.cxx src/Phase8Likelihood.cxx)
 add_executable(phase8_methods apps/phase8_methods.cxx)
 target_link_libraries(phase8_methods PRIVATE atlas_phase8)
 set_target_properties(phase8_methods PROPERTIES BUILD_RPATH "${DELPHES_ROOT}/lib;${ROOT_LIBRARY_DIR}")
endif()
if(BUILD_TESTING)
 foreach(name physics selection histograms statistics likelihood)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_phase8_${name}.cxx")
   add_executable(test_phase8_${name} tests/test_phase8_${name}.cxx)
   target_link_libraries(test_phase8_${name} PRIVATE atlas_phase8)
   add_test(NAME phase8_${name} COMMAND test_phase8_${name})
   set_tests_properties(phase8_${name} PROPERTIES TIMEOUT 120 LABELS "phase8;synthetic_method")
  endif()
 endforeach()
 add_test(NAME phase8_native_methods COMMAND phase8_analyze methods-test)
 set_tests_properties(phase8_native_methods PROPERTIES TIMEOUT 30 LABELS "phase8;synthetic_method")
endif()
