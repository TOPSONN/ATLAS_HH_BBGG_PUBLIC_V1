find_package(ROOT 6.40.04 EXACT CONFIG REQUIRED COMPONENTS Core RIO Tree Hist Graf Gpad MathCore)
if(NOT TARGET atlas_phase8)
  message(FATAL_ERROR "Phase9 reuses the validated Phase8 library")
endif()
foreach(name readback binomial_reference plots)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/apps/phase9_${name}.cxx")
    add_executable(phase9_${name} apps/phase9_${name}.cxx)
    target_link_libraries(phase9_${name} PRIVATE atlas_phase8 ROOT::Graf ROOT::Gpad ROOT::MathCore)
    target_compile_options(phase9_${name} PRIVATE -Wall -Wextra -Wpedantic)
    set_target_properties(phase9_${name} PROPERTIES BUILD_RPATH "${DELPHES_ROOT}/lib;${ROOT_LIBRARY_DIR}")
  endif()
endforeach()
