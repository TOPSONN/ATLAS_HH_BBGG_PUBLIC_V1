cmake_minimum_required(VERSION 3.20)
project(atlas_hh_phase10 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(PHASE10_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
# Reuse the unchanged Phase8 adapter and native regressions; no copied reader.
add_subdirectory("${PHASE10_REPO}/analysis/phase8" "${CMAKE_BINARY_DIR}/phase8")
find_package(ROOT 6.40.04 EXACT CONFIG REQUIRED COMPONENTS Core RIO Tree)
add_executable(phase10_photon_observability source/photon_observability.cxx)
target_link_libraries(phase10_photon_observability PRIVATE atlas_phase8)
target_compile_options(phase10_photon_observability PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(phase10_photon_observability PROPERTIES BUILD_RPATH "${DELPHES_ROOT}/lib;${ROOT_LIBRARY_DIR}")
if(BUILD_TESTING)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_test(NAME phase10_source_methods COMMAND "${Python3_EXECUTABLE}" -B -m unittest discover -s analysis/phase10/tests -p "test_*.py" -v)
 set_tests_properties(phase10_source_methods PROPERTIES WORKING_DIRECTORY "${PHASE10_REPO}" TIMEOUT 90 LABELS "phase10;source;synthetic_method")
endif()
